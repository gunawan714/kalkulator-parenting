function hitungSkor(){


    const usia = document.getElementById("usia").value;

    const reaksi = document.getElementById("reaksi").value;

    const aplikasi = document.getElementById("aplikasi").value;


    let skor = 100;


    if(usia === "6-9") skor -= 5;


    if(reaksi === "tantrum") skor -= 20;

    if(reaksi === "agresif") skor -= 30;

    if(reaksi === "merengek") skor -= 10;


    if(aplikasi === "medsos") skor -= 15;

    if(aplikasi === "belajar") skor += 10;


    skor = Math.max(0, Math.min(100, skor));


    document.getElementById("scoreValue").innerText = skor;


    const warnaCard = document.getElementById("warnaCard");


    let warnaNama = "";

    let warnaHex = "";


    switch(aplikasi){

        case "youtube":

            warnaNama = "Kuning Fokus";

            warnaHex = "#F2C94C";

            break;


        case "game":

            warnaNama = "Ungu Fokus";

            warnaHex = "#8E44AD";

            break;


        case "belajar":

            warnaNama = "Hijau Fokus";

            warnaHex = "#27AE60";

            break;


        case "medsos":

            warnaNama = "Oranye Fokus";

            warnaHex = "#F2994A";

            break;

    }


    warnaCard.style.background = warnaHex;

    warnaCard.innerHTML =

        "Warna Fokus yang Direkomendasikan: <br><strong>" +

        warnaNama + "</strong>";


    const tips = [


        "Buat ritual 3 menit sebelum gadget dimatikan: hitung mundur bersama lalu berikan pelukan atau high-five.",


        "Ajak anak menceritakan satu hal paling menarik yang ditonton hari ini selama 3 menit penuh tanpa interupsi.",


        "Tempelkan jadwal screen time sederhana dan biarkan anak memberi tanda centang setelah berhasil mematuhinya.",


        "Gunakan timer visual dan lakukan aktivitas fisik ringan selama 3 menit setelah layar dimatikan.",


        "Tanyakan tiga pertanyaan reflektif: Apa yang dipelajari? Apa yang menyenangkan? Apa yang ingin dilakukan besok?"

    ];


    let index = 0;


    if(aplikasi === "belajar") index = 4;

    else if(aplikasi === "youtube") index = 1;

    else if(aplikasi === "game") index = 3;

    else if(aplikasi === "medsos") index = 2;


    if(reaksi === "tantrum" || reaksi === "agresif"){

        index = 0;

    }


    document.getElementById("tipsText").innerText = tips[index];


    document.getElementById("hasil").style.display = "block";


    if(skor >= 80){

        confetti();

    }

}


function confetti(){


    const colors = [

        "#F2C94C",

        "#0A2F44",

        "#27AE60",

        "#8E44AD",

        "#F2994A"

    ];


    for(let i=0;i<80;i++){


        const piece = document.createElement("div");


        piece.classList.add("confetti");


        piece.style.left = Math.random()*100 + "vw";

        piece.style.top = "-10px";

        piece.style.background =

            colors[Math.floor(Math.random()*colors.length)];


        piece.style.animationDuration =

            (Math.random()*3+2) + "s";


        document.body.appendChild(piece);


        setTimeout(()=>{

            piece.remove();

        },5000);

    }

}
